terraform {
memory_size = 256


environment {
variables = {
DATA_BUCKET = aws_s3_bucket.datalake.bucket
RAW_PREFIX = "raw/fortnite/"
FORTNITE_API_BASE = var.fortnite_api_base
FORTNITE_API_PATH = var.fortnite_api_path
FORTNITE_API_KEY = var.fortnite_api_key
}
}


source_code_hash = data.archive_file.lambda_zip.output_base64sha256
}


# --- Schedule: EventBridge rule -> Lambda ---
resource "aws_cloudwatch_event_rule" "fetch_schedule" {
name = "${var.project_prefix}-fetch-schedule"
schedule_expression = var.fetch_schedule
}


resource "aws_cloudwatch_event_target" "fetch_target" {
rule = aws_cloudwatch_event_rule.fetch_schedule.name
target_id = "lambda"
arn = aws_lambda_function.fetch_fortnite.arn
}


resource "aws_lambda_permission" "allow_events" {
statement_id = "AllowExecutionFromEventBridge"
action = "lambda:InvokeFunction"
function_name = aws_lambda_function.fetch_fortnite.function_name
principal = "events.amazonaws.com"
source_arn = aws_cloudwatch_event_rule.fetch_schedule.arn
}


# --- Glue: database + crawler (targets S3 raw/) ---
resource "aws_glue_catalog_database" "db" {
name = replace("${var.project_prefix}_db", "-", "_")
}


data "aws_iam_policy_document" "glue_assume" {
statement {
actions = ["sts:AssumeRole"]
principals { type = "Service", identifiers = ["glue.amazonaws.com"] }
}
}


resource "aws_iam_role" "glue_role" {
name = "${var.project_prefix}-glue-role"
assume_role_policy = data.aws_iam_policy_document.glue_assume.json
}


data "aws_iam_policy_document" "glue_policy" {
statement {
actions = ["s3:GetObject", "s3:ListBucket"]
resources = [
aws_s3_bucket.datalake.arn,
"${aws_s3_bucket.datalake.arn}/*"
]
}
statement {
actions = ["logs:CreateLogGroup", "logs:CreateLogStream", "logs:PutLogEvents"]
resources = ["*"]
}
}


resource "aws_iam_policy" "glue_inline" {
name = "${var.project_prefix}-glue-policy"
policy = data.aws_iam_policy_document.glue_policy.json
}


resource "aws_iam_role_policy_attachment" "glue_attach" {
role = aws_iam_role.glue_role.name
policy_arn = aws_iam_policy.glue_inline.arn
}


resource "aws_glue_crawler" "raw_crawler" {
name = "${var.project_prefix}-raw-crawler"
role = aws_iam_role.glue_role.arn
database_name = aws_glue_catalog_database.db.name


s3_target {
path = "s3://${aws_s3_bucket.datalake.bucket}/raw/fortnite/"
}


schedule = "cron(0 1 * * ? *)" # daily @ 01:00 UTC; tweak as needed
}


# --- Athena workgroup (results to S3) ---
resource "aws_athena_workgroup" "wg" {
name = "${var.project_prefix}_wg"
configuration {
enforce_workgroup_configuration = true
result_configuration {
output_location = "s3://${aws_s3_bucket.datalake.bucket}/athena/"
}
}
}
